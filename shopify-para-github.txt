<script>
(function(){
  var root = document.getElementById('section-2');
  if(!root) return;

  function show(){ root.classList.add('is-visible'); }

  var mode = root.getAttribute('data-reveal');

  if(mode === 'load'){
    show();
  } else {
    if(!('IntersectionObserver' in window)){
      show();
      return;
    }
    var io = new IntersectionObserver(function(entries){
      entries.forEach(function(entry){
        if(entry.isIntersecting){
          show();
          io.unobserve(entry.target);
        }
      });
    }, { threshold: 0.2 });
    io.observe(root);
  }

  var wrap = document.getElementById('swapWrap-2');
  if (wrap && window.innerWidth <= 700) {
    wrap.addEventListener('click', function(){ wrap.classList.toggle('swap'); });
  }

  var end = new Date("2026-08-17T23:59:00");
  var d=document.getElementById("cd-days-2"),
      h=document.getElementById("cd-hours-2"),
      m=document.getElementById("cd-mins-2"),
      s=document.getElementById("cd-secs-2");

  function pad(n){ return n.toString().padStart(2,'0'); }

  function tick(){
    var now=new Date(), diff=end-now;
    if(diff < 0) diff=0;

    var sec=Math.floor(diff/1000),
        days=Math.floor(sec/86400),
        hrs=Math.floor((sec%86400)/3600),
        mins=Math.floor((sec%3600)/60),
        secs=sec%60;

    if(d) d.textContent=pad(days);
    if(h) h.textContent=pad(hrs);
    if(m) m.textContent=pad(mins);
    if(s) s.textContent=pad(secs);
  }

  tick();
  setInterval(tick,1000);
})();
</script>

<script>
(function(){
  var section = document.getElementById('wedding-details');
  if (!section) return;

  function show(){ section.classList.add('is-visible'); }

  function isInView(el){
    var r = el.getBoundingClientRect();
    return (r.top < window.innerHeight && r.bottom > 0);
  }

  if (isInView(section)) {
    show();
  }

  if (!('IntersectionObserver' in window)) {
    show();
    return;
  }

  var observer = new IntersectionObserver(function(entries) {
    entries.forEach(function(entry) {
      if (entry.isIntersecting) {
        show();
        observer.unobserve(section);
      }
    });
  }, { threshold: 0.01 });

  observer.observe(section);

  var localTrigger = section.querySelector('.details-local-trigger');
  if (localTrigger) {
    localTrigger.addEventListener('click', function() {
      var nextSection = section.nextElementSibling;
      while (nextSection && nextSection.nodeType === 1) {
        var tag = (nextSection.tagName || '').toLowerCase();
        if (tag !== 'script' && tag !== 'style') break;
        nextSection = nextSection.nextElementSibling;
      }
      if (!nextSection) return;

      var rect = nextSection.getBoundingClientRect();
      var offset = rect.top + window.pageYOffset;

      window.scrollTo({
        top: offset,
        behavior: 'smooth'
      });
    });
  }
})();
</script>

document.addEventListener('DOMContentLoaded', function() {
  var quintaSection = document.getElementById('quinta-falesia');
  var modal = document.getElementById('quintaModal');
  var modalImg = modal ? modal.querySelector('.quinta-modal-img') : null;
  var modalClose = modal ? modal.querySelector('.quinta-modal-close') : null;

  if (!quintaSection) return;

  if ('IntersectionObserver' in window) {
    var observer = new IntersectionObserver(function(entries) {
      entries.forEach(function(entry) {
        if (entry.isIntersecting) {
          quintaSection.classList.add('is-visible');
          observer.unobserve(quintaSection);
        }
      });
    }, { threshold:0.2 });

    observer.observe(quintaSection);
  } else {
    quintaSection.classList.add('is-visible');
  }

  if (modal && modalImg) {
    var thumbs = quintaSection.querySelectorAll('.quinta-grid-item img');

    thumbs.forEach(function(img) {
      img.addEventListener('click', function() {
        modalImg.src = img.src;
        modalImg.alt = img.alt || '';
        modal.classList.add('is-open');
      });
    });

    function closeModal() {
      modal.classList.remove('is-open');
      modalImg.src = '';
      modalImg.alt = '';
    }

    if (modalClose) {
      modalClose.addEventListener('click', function(e) {
        e.stopPropagation();
        closeModal();
      });
    }

    modal.addEventListener('click', function() {
      closeModal();
    });

    var inner = modal.querySelector('.quinta-modal-inner');
    if (inner) {
      inner.addEventListener('click', function(e) {
        e.stopPropagation();
      });
    }

    document.addEventListener('keydown', function(e) {
      if (e.key === 'Escape' && modal.classList.contains('is-open')) {
        closeModal();
      }
    });
  }
});
</script>

<script>
document.addEventListener('DOMContentLoaded', function() {
  var section = document.getElementById('accommodation-section');
  var carousel = document.getElementById('accCarousel');
  if (!section || !carousel) return;

  var slides = carousel.querySelectorAll('.acc-slide');
  var total = slides.length;
  var index = 0;

  function goTo(i) {
    index = (i + total) % total;
    carousel.style.transform = 'translateX(' + (-index * 100) + '%)';
  }

  var prevBtn = section.querySelector('.acc-arrow-left');
  var nextBtn = section.querySelector('.acc-arrow-right');

  if (prevBtn && nextBtn) {
    prevBtn.addEventListener('click', function() { goTo(index - 1); });
    nextBtn.addEventListener('click', function() { goTo(index + 1); });
  }

  /* swipe mobile */
  var startX = null;
  var threshold = 40;

  carousel.addEventListener('touchstart', function(e) {
    if (e.touches && e.touches.length === 1) {
      startX = e.touches[0].clientX;
    }
  });

  carousel.addEventListener('touchend', function(e) {
    if (startX === null) return;
    var diff = e.changedTouches[0].clientX - startX;
    if (diff > threshold) goTo(index - 1);
    else if (diff < -threshold) goTo(index + 1);
    startX = null;
  });

  /* scroll animation */
  if ('IntersectionObserver' in window) {
    var observer = new IntersectionObserver(function(entries) {
      entries.forEach(function(entry) {
        if (entry.isIntersecting) {
          section.classList.add('is-visible');
          observer.unobserve(section);
        }
      });
    }, { threshold:0.2 });
    observer.observe(section);
  } else {
    section.classList.add('is-visible');
  }
});
</script>
